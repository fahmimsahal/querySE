WITH asg AS (          -- >>> CAKUPAN: sama dengan query_geotag, per kecamatan <<<
    SELECT a.assignment_id, a.assignment_date_modified, a.survey_period_id,
           a.assignment_status_alias, a.mode, a.code_identity, a.data1, a.data6,
           a.level_3_full_code, a.level_3_name, a.level_4_full_code, a.level_4_name,
           a.level_5_full_code, a.level_5_name, a.level_6_full_code, a.level_6_name
    FROM tgr_fd68e454.base_table_assignment a
    WHERE a.level_2_full_code = '{{KAB}}'
      AND a.level_3_full_code = '{{KEC}}'
      AND a.is_active = 1
      AND UPPER(COALESCE(a.assignment_status_alias, '')) NOT IN ('OPEN', 'DRAFT')
),
rv AS (           -- versi data terbaru root_table per assignment
    SELECT r.assignment_id, MAX(r.assignment_date_modified) AS adm
    FROM tgr_fd68e454.root_table r
    INNER JOIN asg g ON g.assignment_id = r.assignment_id
    GROUP BY r.assignment_id
),
xv AS (           -- versi data terbaru nested_dtsen_var per assignment
    SELECT v.assignment_id, MAX(v.assignment_date_modified) AS adm
    FROM tgr_fd68e454.nested_dtsen_var v
    INNER JOIN asg g ON g.assignment_id = v.assignment_id
    GROUP BY v.assignment_id
)
SELECT
/*KOLOM*/
FROM asg a
INNER JOIN rv
        ON rv.assignment_id = a.assignment_id
INNER JOIN tgr_fd68e454.root_table r
        ON r.assignment_id = rv.assignment_id
       AND r.assignment_date_modified = rv.adm
INNER JOIN xv
        ON xv.assignment_id = a.assignment_id
INNER JOIN tgr_fd68e454.nested_dtsen_var v
        ON v.assignment_id = xv.assignment_id
       AND v.assignment_date_modified = xv.adm
WHERE NULLIF(TRIM(r.geotag_latitude), '')  IS NOT NULL    -- syarat geotag (sama dengan query_geotag)
  AND NULLIF(TRIM(r.geotag_longitude), '') IS NOT NULL
ORDER BY a.assignment_id, v.index1
LIMIT 9000 OFFSET 0
