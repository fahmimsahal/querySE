-- ---------------------------------------------------------------------
-- FORMAT FILE (dibaca userscript "Tarik Data Lebar SE2026"):
--   -- @KUNCI  : kolom kunci penggabung, ikut di SETIAP grup query.
--   -- @KOLOM  : satu baris = satu kolom (ekspresi + AS alias). Script memecahnya
--                jadi grup berisi (25 - jumlah kunci) kolom. Baris diawali '--'
--                diabaikan, jadi kolom bisa dinonaktifkan dengan menambah '-- '.
--   -- @QUERY  : template query. /*KOLOM*/ diganti kunci + kolom grup,
--                {{KAB}} dan {{KEC}} diganti kode wilayah dari popup.
-- ATURAN: alias tidak boleh kembar; SELECT terluar tanpa GROUP BY/DISTINCT;
--         ORDER BY terakhir harus unik & diikuti LIMIT 9000 OFFSET 0.
-- ---------------------------------------------------------------------

-- @KUNCI
a.assignment_id AS `assignment_id`
CONCAT('', d.index1) AS `index1`

-- @KOLOM

-- [IDENTITAS ASSIGNMENT]
a.`level_3_full_code` AS `level_3_full_code`   -- kode kecamatan
a.`level_4_full_code` AS `level_4_full_code`   -- kode desa
a.`level_4_name` AS `level_4_name`
a.`level_6_full_code` AS `level_6_full_code`   -- kode sub-SLS
a.`level_6_name` AS `level_6_name`
a.`assignment_status_alias` AS `assignment_status_alias`
r.`nama_kk` AS `nama_kk_root`   -- nama KK dari root_table (bantu baca)
CONCAT('https://fasih-sm.bps.go.id/app/assignment/', a.survey_period_id, '/', a.assignment_id) AS `link_fasih`
d.`assignment_date_modified` AS `versi_roster`

-- ===================== nested_dtsen (d.) =====================

-- [SE2026 - L BLOK I]
d.`nama_dtsen` AS `nama_dtsen`   -- Nama anggota keluarga
d.`nik_dtsen` AS `nik_dtsen`   -- Nomor Induk Kependudukan (NIK) NAME (Tulis sesuai yang tercantum di KK…
d.`hubungan_label` AS `hubungan_label`   -- Hubungan NAME dengan Kepala Keluarga
d.`hubungan_value` AS `hubungan_value`   -- Hubungan NAME dengan Kepala Keluarga
d.`keberadaan_dtsen_label` AS `keberadaan_dtsen_label`   -- a. Keberadaan (status) anggota keluarga saat ini
d.`keberadaan_dtsen_value` AS `keberadaan_dtsen_value`   -- a. Keberadaan (status) anggota keluarga saat ini
d.`alamat_dn_label` AS `alamat_dn_label`   -- b. Alamat domisili:
d.`alamat_dn_value` AS `alamat_dn_value`   -- b. Alamat domisili:
d.`prov_dn_label` AS `prov_dn_label`   -- Provinsi
d.`prov_dn_value` AS `prov_dn_value`   -- Provinsi
d.`kab_dn_label` AS `kab_dn_label`   -- Kabupaten/Kota
d.`kab_dn_value` AS `kab_dn_value`   -- Kabupaten/Kota
d.`domisili_ln_label` AS `domisili_ln_label`   -- Negara domisili Individu (luar negeri)
d.`domisili_ln_value` AS `domisili_ln_value`   -- Negara domisili Individu (luar negeri)
d.`status_kawin_label` AS `status_kawin_label`   -- Apakah status perkawinan NAME
d.`status_kawin_value` AS `status_kawin_value`   -- Apakah status perkawinan NAME

-- [LAINNYA (variabel sistem kuesioner)]
d.`isprelistart` AS `isprelistart`   -- Apakah prelist art

d.`nik_dtsen_prelist` AS `nik_dtsen_prelist`   -- nik_dtsen_prelist

d.`jk_prelist` AS `jk_prelist`   -- jk_prelist

-- @QUERY
WITH asg AS (          -- >>> CAKUPAN: sama dengan query_geotag, per kecamatan <<<
    SELECT a.assignment_id, a.assignment_date_modified, a.survey_period_id,
           a.assignment_status_alias, a.mode, a.code_identity, a.data1, a.data6,
           a.level_3_full_code, a.level_3_name, a.level_4_full_code, a.level_4_name,
           a.level_5_full_code, a.level_5_name, a.level_6_full_code, a.level_6_name
    FROM tgr_fd68e454.base_table_assignment a
    WHERE a.level_2_full_code = '{{KAB}}'
      AND a.level_3_full_code = '{{KEC}}'
      AND a.is_active = 1
      AND UPPER(COALESCE(a.assignment_status_alias, '')) NOT IN ('OPEN', 'DRAFT')
),
rv AS (           -- versi data terbaru root_table per assignment
    SELECT r.assignment_id, MAX(r.assignment_date_modified) AS adm
    FROM tgr_fd68e454.root_table r
    INNER JOIN asg g ON g.assignment_id = r.assignment_id
    GROUP BY r.assignment_id
),
xv AS (           -- versi data terbaru nested_dtsen per assignment
    SELECT d.assignment_id, MAX(d.assignment_date_modified) AS adm
    FROM tgr_fd68e454.nested_dtsen d
    INNER JOIN asg g ON g.assignment_id = d.assignment_id
    GROUP BY d.assignment_id
)
SELECT
/*KOLOM*/
FROM asg a
INNER JOIN rv
        ON rv.assignment_id = a.assignment_id
INNER JOIN tgr_fd68e454.root_table r
        ON r.assignment_id = rv.assignment_id
       AND r.assignment_date_modified = rv.adm
INNER JOIN xv
        ON xv.assignment_id = a.assignment_id
INNER JOIN tgr_fd68e454.nested_dtsen d
        ON d.assignment_id = xv.assignment_id
       AND d.assignment_date_modified = xv.adm
WHERE NULLIF(TRIM(r.geotag_latitude), '')  IS NOT NULL    -- syarat geotag (sama dengan query_geotag)
  AND NULLIF(TRIM(r.geotag_longitude), '') IS NOT NULL
ORDER BY a.assignment_id, d.index1
LIMIT 9000 OFFSET 0
